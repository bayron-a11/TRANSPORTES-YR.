    <h2>Registro de Usuario</h2>
    <input type="text" id="nombre" placeholder="Nombre completo">
    <select id="tipoId">
      <option value="CC">Cédula de ciudadanía</option>
      <option value="TI">Tarjeta de identidad</option>
      <option value="CE">Cédula de extranjería</option>
    </select>
    <input type="text" id="numeroId" placeholder="Número de identificación">
    <input type="date" id="fechaNacimiento">
    <input type="text" id="usuario" placeholder="Nombre de usuario">
    <input type="password" id="clave" placeholder="Contraseña">
    <button onclick="registrarUsuario()">Registrar</button>
    <button onclick="validarCedula()">Validar cédula</button>
    <div class="mensaje" id="mensajeCedula"></div>
    <button onclick="borrarUsuarios()" style="background-color: crimson;">Vaciar memoria</button>
  </div>

  <script>
    const fondo = document.getElementById('fondoImagen');
    const video = document.getElementById('videoFondo');
    const claxon = document.getElementById('claxon');

    const imagenes = [
      'imagenes/foto1.png',
      'imagenes/foto2.png',
      'imagenes/foto3.png'
    ];

    let i = 0;

    function mostrarFondo() {
      if (i < imagenes.length) {
        fondo.src = imagenes[i];
        i++;
        setTimeout(mostrarFondo, 3000);
      } else {
        fondo.style.display = 'none';
        video.style.display = 'block';
        video.play();
        claxon.play();
        video.onended = () => {
          video.style.display = 'none';
          fondo.style.display = 'block';
          i = 0;
          mostrarFondo();
        };
      }
    }

    window.onload = mostrarFondo;

    function registrarUsuario() {
      const usuarios = JSON.parse(localStorage.getItem('usuarios')) || [];
      const nuevoUsuario = {
        nombre: document.getElementById("nombre").value,
        tipoId: document.getElementById("tipoId").value,
        numeroId: document.getElementById("numeroId").value,
        fechaNacimiento: document.getElementById("fechaNacimiento").value,
        usuario: document.getElementById("usuario").value,
        clave: document.getElementById("clave").value
      };
      const existe = usuarios.some(u => u.usuario === nuevoUsuario.usuario);
      if (existe) {
        alert('El nombre de usuario ya está registrado.');
        return;
      }
      usuarios.push(nuevoUsuario);
      localStorage.setItem('usuarios', JSON.stringify(usuarios));
      alert("Registro exitoso. Ahora valida tu cédula.");
    }

    function validarCedula() {
      const cedula = document.getElementById("numeroId").value.trim();
      const usuarios = JSON.parse(localStorage.getItem('usuarios')) || [];
      const existe = usuarios.some(u => u.numeroId === cedula);
      if (existe) {
        alert("Cédula validada. Bienvenido.");
        window.location.href = "unificado.html";
      } else {
        document.getElementById("mensajeCedula").innerText = "Cédula no registrada.";
      }
    }

    function borrarUsuarios() {
      localStorage.removeItem('usuarios');
      alert("Usuarios eliminados.");
    }
  </script>
